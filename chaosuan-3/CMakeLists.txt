cmake_minimum_required(VERSION 3.16)
project(chaosuan3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
function(configure_filter target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
    endif()
endfunction()
add_executable(log_filter main.cpp filter.cpp)
configure_filter(log_filter)
add_executable(baseline_refactored main.cpp serial_filter.cpp)
target_compile_definitions(baseline_refactored PRIVATE FILTER_IMPLEMENTATION="serial_refactored")
configure_filter(baseline_refactored)
add_executable(baseline_original baseline.cpp)
configure_filter(baseline_original)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_api test_api.cpp filter.cpp)
    configure_filter(test_api)
    add_test(NAME api COMMAND test_api)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME public COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_public.py $<TARGET_FILE:log_filter>)
        add_test(NAME regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_regression.py $<TARGET_FILE:log_filter>)
    endif()
endif()
